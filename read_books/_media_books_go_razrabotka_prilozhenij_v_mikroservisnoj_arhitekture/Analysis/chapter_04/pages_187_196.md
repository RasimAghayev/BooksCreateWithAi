func (r *Repository) GetBalance(ctx context.Context, userID uint64) (float32, error) {
    var user repomodel.User
    res := r.db.WithContext(ctx).
        Model(&repomodel.User{}).
        Select("balance").
        Where("id = ? AND is_deleted = ?", userID, false).
        First(&user)
    if errors.Is(res.Error, gorm.ErrRecordNotFound) {
        return 0, fmt.Errorf("user not found")
    } else if res.Error != nil {
        return 0, fmt.Errorf("failed to get user balance: %w", res.Error)
    }
    return user.Balance, nil
}

#### UpdateBalance (listing 4.17)
```go
func (r *Repository) UpdateBalance(ctx context.Context, userID uint64, amount float32, operationType model.BalanceOperationType) (float32, float32, error) {
    var user repomodel.User
    // 1. Get current balance
    res := r.db.WithContext(ctx).
        Model(&repomodel.User{}).
        Where("id = ? AND is_deleted = ?", userID, false).
        First(&user)
    if errors.Is(res.Error, gorm.ErrRecordNotFound) {
        return 0, 0, fmt.Errorf("user not found")
    } else if res.Error != nil {
        return 0, 0, fmt.Errorf("failed to get user for balance update: %w", res.Error)
    }
    
    oldBalance := user.Balance
    var newBalance float32
    
    // 2. Calculate new balance based on operation type
    switch operationType {
    case model.BalanceOperationDeposit:
        newBalance = oldBalance + amount
    case model.BalanceOperationCredit:
        newBalance = oldBalance - amount
        if newBalance < 0 {
            return 0, 0, fmt.Errorf("insufficient balance: current balance %.2f, requested amount %.2f", oldBalance, amount)
        }
    default:
        return 0, 0, fmt.Errorf("invalid balance operation type: %s", operationType)
    }
    
    // 3. Update in database
    res = r.db.WithContext(ctx).
        Model(&repomodel.User{}).
        Where("id = ? AND is_deleted = ?", userID, false).
        Update("balance", newBalance)
    if res.Error != nil {
        return 0, 0, fmt.Errorf("failed to update user balance: %w", res.Error)
    }
    return oldBalance, newBalance, nil
}
```

#### TransferBalance (listing 4.17)
```go
func (r *Repository) TransferBalance(ctx context.Context, fromUserID, toUserID uint64, amount float32) error {
    if amount <= 0 {
        return fmt.Errorf("transfer amount must be positive")
    }
    if fromUserID == toUserID {
        return fmt.Errorf("cannot transfer to the same user")
    }
    
    return r.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
        // 1. Verify recipient exists
        var toUser repomodel.User
        res := tx.Model(&repomodel.User{}).
            Where("id = ? AND is_deleted = ?", toUserID, false).
            First(&toUser)
        if errors.Is(res.Error, gorm.ErrRecordNotFound) {
            return fmt.Errorf("recipient user not found")
        }
        
        // 2. Verify sender exists and has sufficient funds
        var fromUser repomodel.User
        res = tx.Model(&repomodel.User{}).
            Where("id = ? AND is_deleted = ?", fromUserID, false).
            First(&fromUser)
        if errors.Is(res.Error, gorm.ErrRecordNotFound) {
            return fmt.Errorf("sender user not found")
        }
        if fromUser.Balance < amount {
            return fmt.Errorf("insufficient balance: current balance %.2f, transfer amount %.2f", fromUser.Balance, amount)
        }
        
        // 3. Deduct from sender
        res = tx.Model(&repomodel.User{}).
            Where("id = ? AND is_deleted = ?", fromUserID, false).
            Update("balance", fromUser.Balance-amount)
        if res.Error != nil {
            return fmt.Errorf("failed to deduct from sender balance: %w", res.Error)
        }
        
        // 4. Add to recipient
        res = tx.Model(&repomodel.User{}).
            Where("id = ? AND is_deleted = ?", toUserID, false).
            Update("balance", toUser.Balance+amount)
        if res.Error != nil {
            return fmt.Errorf("failed to add to recipient balance: %w", res.Error)
        }
        return nil
    })
}
```

### Balance Operation Types

The `BalanceOperationType` is defined as an enum in the model:
- **DEPOSIT**: Increases balance (credit to user)
- **CREDIT**: Decreases balance (debit from user)
- **WITHDRAW**: Alias for credit operation
- Note: This is **accounting credit/debit from the user's perspective**, not double-entry bookkeeping. When a deposit is made, the user's account is CREDITED (increases). When a withdrawal is made, the user's account is DEBITED (decreases). However, in the TransactionEntry model, Deposit creates a DEBIT entry and Withdraw creates a CREDIT entry — this is the double-entry perspective.

### Account Service Layer Balance Methods (listing 4.18)

```go
type AccountService struct {
    repo   Repository
    logger *zerolog.Logger
}

type Repository interface {
    // ... existing methods (CreateUser, GetUser, etc.)
    GetBalance(context.Context, uint64) (float32, error)
    UpdateBalance(context.Context, uint64, float32, model.BalanceOperationType) (float32, float32, error)
    TransferBalance(context.Context, uint64, uint64, float32) error
}

func (s *AccountService) GetBalance(ctx context.Context, userID uint64) (model.GetBalanceResponse, error) {
    balance, err := s.repo.GetBalance(ctx, userID)
    if err != nil {
        return model.GetBalanceResponse{}, err
    }
    return model.GetBalanceResponse{UserID: userID, Balance: balance}, nil
}

func (s *AccountService) UpdateBalance(ctx context.Context, req model.UpdateBalanceRequest) (model.UpdateBalanceResponse, error) {
    // Validation
    if req.Amount < 0 {
        return model.UpdateBalanceResponse{}, fmt.Errorf("amount cannot be negative")
    }
    if req.Type == model.BalanceOperationCredit && req.Amount == 0 {
        return model.UpdateBalanceResponse{}, fmt.Errorf("amount must be positive for subtract operation")
    }
    
    oldBalance, newBalance, err := s.repo.UpdateBalance(ctx, req.UserID, req.Amount, req.Type)
    if err != nil {
        return model.UpdateBalanceResponse{}, err
    }
    return model.UpdateBalanceResponse{
        UserID: req.UserID,
        OldBalance: oldBalance,
        NewBalance: newBalance,
        Amount: req.Amount,
        Operation: req.Type,
    }, nil
}

func (s *AccountService) TransferBalance(ctx context.Context, fromUserID, toUserID uint64, amount float32) error {
    if amount <= 0 {
        return fmt.Errorf("transfer amount must be positive")
    }
    if fromUserID == toUserID {
        return fmt.Errorf("cannot transfer to the same user")
    }
    err := s.repo.TransferBalance(ctx, fromUserID, toUserID, amount)
    if err != nil {
        return err
    }
    return nil
}
```

## What the Author Is Saying

> "Метод updateBalance инкапсулирует логику изменения средств в зависимости от типа операции — пополнение или списание, при этом сразу проверяется достаточность средств на счете, чтобы предотвратить уход в отрицательный баланс."

The balance update must check for sufficient funds BEFORE debiting.

> "Для перевода средств добавлен метод TransferBalance, который содержит более сложную логику — в рамках одной транзакции базы данных выполняется перевод средств от одного пользователя к другому."

TransferBalance is atomic — both sender debit and recipient credit happen together, or both fail.

## Important Concepts

- **Balance validation**: Always check `newBalance >= 0` before allowing a credit (withdrawal) operation.
- **Transfer atomicity**: Both debit and credit happen in a single DB transaction — if either fails, both roll back.
- **BalanceOperationType**: DEPOSIT (increase) and CREDIT (decrease) — note this is account-level, not transaction-entry-level.
- **Service layer validation**: Additional input validation in the service layer (amount > 0, not same user) before calling repository.
- **Error responses**: TransferBalance returns an error (not details) — the caller only needs success/failure.
- **Old vs new balance returned**: UpdateBalance returns both old and new values for logging and audit purposes.

## Remember This

> Balance updates check for sufficient funds before debiting — `if newBalance < 0 { return error("insufficient balance") }`.

> Transfers are atomic within a single GORM `Transaction()` — both updates succeed or both fail.

## Terms

- **UpdateBalance**: Repository method for deposit/credit with validation
- **TransferBalance**: Atomic transfer between two user accounts
- **BalanceOperationType**: Enum for deposit vs credit (DEPOSIT, CREDIT)
- **Insufficient balance**: Error returned when withdrawal would make balance negative
- **OldBalance / NewBalance**: Returned together for audit/logging
- **Atomic transfer**: Both debit+credit in single DB transaction

## Notes

- The BalanceOperationType in the model layer (DEPOSIT/CREDIT) uses different terminology than TransactionEntry's Direction (DEBIT/CREDIT). This is a naming inconsistency in the book.
- The `UpdateBalance` method returns (float32, float32, error) — old balance, new balance, and error. This is useful for logging and response.
- The `TransferBalance` method in the service layer performs additional validation (positive amount, not same user) before calling the repository.
- The service layer wraps repository calls with structured logging using zerolog.
