func GetBookHandler(w http.ResponseWriter, r *http.Request) {
   vars := mux.Vars(r)
   bookID := vars["id"]
   var fetchedBook Book
    if err := BookstoreDB.First(&fetchedBook, bookID).Error; err
!= nil {
       http.Error(w, "Book not found", http.StatusNotFound)
       return
   }
   json.NewEncoder(w).Encode(fetchedBook)
}
Given a book ID, this handler retrieves and displays the respective book's
details.
