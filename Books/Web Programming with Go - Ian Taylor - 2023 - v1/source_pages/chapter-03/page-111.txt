       return
   }
   decoder := json.NewDecoder(r.Body)
   err := decoder.Decode(&updatedBook)
   if err != nil {
       http.Error(w, "Invalid data", http.StatusBadRequest)
       return
   }
   BookstoreDB.Save(&updatedBook)
   w.WriteHeader(http.StatusOK)
   json.NewEncoder(w).Encode(updatedBook)
}
