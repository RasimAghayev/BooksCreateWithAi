Suppose there's a third-party service providing book summaries. You could
fetch a book's summary like:
func GetBookSummary(bookID string) (string, error) {
   url := fmt.Sprintf("https://booksummaryapi.com/summary/%s",
bookID)
   resp, err := http.Get(url)
   if err != nil {
       return "", err
   }
   defer resp.Body.Close()
   body, _ := ioutil.ReadAll(resp.Body)
   return string(body), nil
}
