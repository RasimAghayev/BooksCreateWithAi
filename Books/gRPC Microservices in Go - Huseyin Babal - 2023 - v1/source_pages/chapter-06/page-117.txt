    // handle caCertErr
    if ok := certPool.AppendCertsFromPEM(caCert); !ok { 
        return nil, errors.New("failed to append the CA certs")
    }
    return credentials.NewTLS(
        &tls.Config{
            ClientAuth:   tls.RequireAnyClientCert,
            Certificates: []tls.Certificate{serverCert}, 
            ClientCAs:    certPool, 
        }), nil
}
