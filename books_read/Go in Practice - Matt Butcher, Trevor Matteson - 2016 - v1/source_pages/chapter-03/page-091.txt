Concurrency in Go
The following listing is an example of a program with a race condition. This simple
program reads any number of files and tallies the number of occurrences for each
word it finds. At the end of its execution, it prints a list of words that appear more
than once.
package main
import (
     "bufio"
     "fmt"
     "os"
     "strings"
     "sync"
)
func main() {
     var wg sync.WaitGroup 
     w := newWords()
     for _, f := range os.Args[1:] {
            wg.Add(1)
            go func(file string) {
                   if err := tallyWords(file, w); err != nil {
                           fmt.Println(err.Error())
                   }
                   wg.Done()
            }(f)
     }
     wg.Wait()
     fmt.Println("Words that appear more than once:")
     for word, count := range w.found {
            if count > 1 {
                   fmt.Printf("%s: %d\n", word, count)
            }
     }
}
type words struct {
     found map[string]int
}
func newWords() *words {
     return &words{found: map[string]int{}}
}
