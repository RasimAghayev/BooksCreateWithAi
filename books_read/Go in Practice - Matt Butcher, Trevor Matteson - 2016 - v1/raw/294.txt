func fieldName(field reflect.StructField) string {
     if t := field.Tag.Get("ini"); t != "" {
            return t
     }
     return field.Name 
}
func Marshal(v interface{}) ([]byte, error) {
     var b bytes.Buffer
     val := 
reflect.Indirect(reflect.ValueOf(v))
     if val.Kind() != reflect.Struct {
            return []byte{}, errors.New("unmarshal can only take structs")
     }
     t := val.Type()
     for i := 0; i < t.NumField(); i++ {
            f := t.Field(i)
            name := fieldName(f)
            raw := val.Field(i).Interface()
            fmt.Fprintf(&b, "%s=%v\n", name, raw)
     }
     return b.Bytes(), nil
}
This Marshal() function takes the given v interface{} and reads through its fields.
By examining the type, it can iterate through all the fields on the struct, and for each
field, it can access the annotation (via StructField.Tag()). As it loops through the
struct fields, it can also fetch the relevant values for each struct field. Rather than man-
ually convert these values from their native type to a string, you rely on fmt.Fprintf()
to do that work for you.
 Of note, the fieldName() function uses Go’s automatic tag parsing. Although you
can (if you desire) store any string data in an annotation, Go can parse tags for you.
