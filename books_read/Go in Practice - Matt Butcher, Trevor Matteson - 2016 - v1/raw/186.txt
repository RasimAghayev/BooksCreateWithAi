The index.html file is a template for the page wrapper. It contains variables that make
sense in the scope of a page. The variables printed out aren’t specific to a user or any-
thing else. The following listing pulls this together.
package main
import (
     "bytes"
     "html/template"
     "net/http"
)
var t *template.Template
var qc template.HTML
func init() {
     t = template.Must(template.ParseFiles("index.html", "quote.html"))
}
type Page struct {
     Title   string
     Content template.HTML
}
type Quote struct {
     Quote, Name string
}
func main() {
     q := &Quote{
            Quote: `You keep using that word. I do not think
                    it means what you think it means.`,
            Person: "Inigo Montoya",
     }
     var b bytes.Buffer
     t.ExecuteTemplate(&b, "quote.html", q)
