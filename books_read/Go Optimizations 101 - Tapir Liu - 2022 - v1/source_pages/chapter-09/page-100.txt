//go:noinline
func Concat_WithPlus(a, b, c, d string) string {
return a + b + c + d
}
//go:noinline
func Concat_WithBytes(ss ...string) string {
var n = 0
for _, s := range ss {
n += len(s)
}
var bs []byte
if n > 64 {
bs = make([]byte, 0, n) // escapes to heap
} else {
bs = make([]byte, 0, 64) // does not escape
}
for _, s := range ss {
bs = append(bs, s...)
}
return string(bs)
}
