a[i * 4 + 0] = byte(v >> 0)
a[i * 4 + 1] = byte(v >> 8)
a[i * 4 + 2] = byte(v >> 16)
a[i * 4 + 3] = byte(v >> 24)
}
var v = a[0]
for i := 1; i < len(a); i++ {
r += v ^ a[i]
v = a[i]
}
return
}
func foo(n byte) uint16 {
var a [2500]uint32
for i := range a {
a[i] = rand.Uint32()
}
x := bar(a[:1250], n)
y := bar(a[1250:], ^n)
return uint16(x) << 8 | uint16(y)
}
func main() {
x := foo(123)
fmt.Println(x)
duck()
}
