x[j][k] = new(int)
}
}
time.Sleep(time.Second / 1000)
}
}
func bigStack(c chan int, v byte) byte {
defer func() {
c <- 1
}()
var s = []byte{150 << 20: 0} // on stack
for i := range s { s[i] = v }
return s[v]
}
func main() {
go bigStack(nil, 123)
garbageProducer() // never exit
}
