// ReportHealthyState is a push mechanism for
// reporting healthy state to the registry.
func (r *Registry) ReportHealthyState(instanceID string, 
serviceName string) error {
    r.Lock()
    defer r.Unlock()
    if _, ok := r.serviceAddrs[serviceName]; !ok {
        return errors.New("service is not registered 
yet")
    }
    if _, ok := r.serviceAddrs[serviceName][instanceID]; 
!ok {
        return errors.New("service instance is not 
registered yet")
    }
    r.serviceAddrs[serviceName][instanceID].lastActive = 
time.Now()
    return nil
}
