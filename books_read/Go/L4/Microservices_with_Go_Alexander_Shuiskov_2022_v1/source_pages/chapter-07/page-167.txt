// Put adds a rating for a given record.
func (r *Repository) Put(ctx context.Context, recordID model.
RecordID, recordType model.RecordType, rating *model.Rating) 
error {
    _, err := r.db.ExecContext(ctx, "INSERT INTO ratings 
(record_id, record_type, user_id, value) VALUES (?, ?, ?, ?)",
        recordID, recordType, rating.UserID, rating.Value)
    return err
}
