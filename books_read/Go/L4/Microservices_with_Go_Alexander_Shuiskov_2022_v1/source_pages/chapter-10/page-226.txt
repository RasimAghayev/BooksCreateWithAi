Here is an example of using a token bucket-based rate limiter in Go:
package main
import (
    "fmt"
    "golang.org/x/time/rate"
)
func main() {
    limit := 3
    burst := 3
    limiter := rate.NewLimiter(rate.Limit(limit), burst)
    for i := 0; i < 100; i++ {
        if limiter.Allow() {
            fmt.Println("allowed")
        } else {
            fmt.Println("not allowed")
        }
    }
}
This code prints allowed 3 times and then keeps printing not allowed 97 times unless it takes 
more than 1 second to run.
