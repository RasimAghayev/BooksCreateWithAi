// ServiceAddresses returns the list of addresses of
// active instances of the given service.
func (r *Registry) ServiceAddresses(ctx context.Context, 
serviceName string) ([]string, error) {
    entries, _, err := r.client.Health().
Service(serviceName, "", true, nil)
    if err != nil {
        return nil, err
    } else if len(entries) == 0 {
        return nil, discovery.ErrNotFound
    }
    var res []string
    for _, e := range entries {
        res = append(res, res = append(res, fmt.
Sprintf("%s:%d", e.Service.Address, e.Service.Port)))
    }
    return res, nil
}
