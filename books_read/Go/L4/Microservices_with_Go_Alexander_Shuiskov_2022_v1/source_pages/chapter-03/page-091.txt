// NewRegistry creates a new Consul-based service
// registry instance.
func NewRegistry(addr string) (*Registry, error) {
    config := consul.DefaultConfig()
    config.Address = addr
    client, err := consul.NewClient(config)
    if err != nil {
        return nil, err
    }
    return &Registry{client: client}, nil
}
